import ctypes, json, mmap, os, re, shlex, signal, subprocess, sys

PS = mmap.PAGESIZE
LOWBIT = bytes(b & 1 for b in range(256))
THP = "/sys/kernel/mm/transparent_hugepage/"
VM = ("dirty_ratio", "dirty_background_ratio", "dirty_bytes", "dirty_background_bytes",
      "dirty_expire_centisecs", "dirty_writeback_centisecs", "swappiness", "vfs_cache_pressure")

class StepFailed(Exception):
    pass


def why(e):
    return "%s: %s" % (type(e).__name__, " ".join(str(e).split()))


def run(argv):
    """stdout of argv, or None when it is missing, fails or takes over 10 s."""
    try:
        p = subprocess.run(argv, stdin=subprocess.DEVNULL, stdout=subprocess.PIPE, stderr=subprocess.DEVNULL,
                           timeout=10, universal_newlines=True, errors="replace")
    except (OSError, subprocess.SubprocessError):
        return None
    return p.stdout if p.returncode == 0 else None


def text(path):
    try:
        with open(path, errors="replace") as f:
            return f.read().strip()
    except OSError:
        return None


def number(path):
    t = text(path)
    return int(t) if t is not None and re.match(r"^[0-9]+$", t) else None


def active(path):
    """The active entry of a sysfs choice file ("always [madvise] never" -> "madvise"), else its text."""
    t = text(path)
    m = re.search(r"\[([^]]*)\]", t or "")
    return m.group(1) if m else t


def key_value(path, key, sep):
    """The value of the first `<key><sep>` line of <path>, shell quotes removed."""
    for line in (text(path) or "").splitlines():
        k, s, v = line.partition(sep)
        if s and k.strip() == key:
            return " ".join(shlex.split(v))
    return None


def mem_total_kb():
    for line in (text("/proc/meminfo") or "").splitlines():
        if line.startswith("MemTotal:"):
            return int(line.split()[1])
    return None


def libc():
    if not _libc:
        c = ctypes.CDLL(None, use_errno=True)
        c.mmap.restype = ctypes.c_void_p
        c.mmap.argtypes = (ctypes.c_void_p, ctypes.c_size_t, ctypes.c_int, ctypes.c_int, ctypes.c_int, ctypes.c_long)
        c.munmap.argtypes = (ctypes.c_void_p, ctypes.c_size_t)
        c.mincore.argtypes = (ctypes.c_void_p, ctypes.c_size_t, ctypes.c_void_p)
        _libc.append(c)
    return _libc[0]


class Mapped(object):
    """<path> mapped shared and read-only. Mapping faults no page in, and nothing reads the map."""

    def __init__(self, path):
        self.addr = None
        self.fd = os.open(path, os.O_RDONLY)
        self.size = os.fstat(self.fd).st_size
        self.pages = (self.size + PS - 1) // PS
        if self.size:
            c = libc()
            a = c.mmap(None, self.size, mmap.PROT_READ, mmap.MAP_SHARED, self.fd, 0)
            if not a or a == ctypes.c_void_p(-1).value:
                e = ctypes.get_errno()
                self.close()
                raise OSError(e, "mmap: " + os.strerror(e))
            self.addr = a

    def resident(self):
        """mincore of the map: one byte per page, 1 when the page cache holds the page."""
        vec = (ctypes.c_ubyte * max(self.pages, 1))()
        if self.pages and libc().mincore(self.addr, self.size, vec) != 0:
            e = ctypes.get_errno()
            raise OSError(e, "mincore: " + os.strerror(e))
        return bytes(vec)[:self.pages].translate(LOWBIT)

    def close(self):
        if self.addr:
            libc().munmap(self.addr, self.size)
            self.addr = None
        if self.fd is not None:
            os.close(self.fd)
            self.fd = None


def resident_pages(path):
    """Resident pages of <path>: fincore's count when fincore runs, else mincore over an untouched map."""
    out = (run(["fincore", "--noheadings", "--raw", "--output", "PAGES", path]) or "").strip()
    if re.match(r"^[0-9]+$", out):
        return int(out)
    m = Mapped(path)
    try:
        return m.resident().count(1)
    finally:
        m.close()


def mount_of(d):
    """(source, fstype, options) of the mount holding <d>, from findmnt."""
    try:
        fs = json.loads(run(["findmnt", "--json", "--output", "SOURCE,FSTYPE,OPTIONS", "--target", d]))
        fs = fs["filesystems"][0]
        return fs.get("source"), fs.get("fstype"), fs.get("options")
    except (TypeError, ValueError, KeyError, IndexError):
        return None, None, None


def data_disk(d, source):
    """The block device whose queue serves <d>: lsblk's PKNAME of the mount source (the disk under a
    partition), else its KNAME (a whole disk, a device-mapper volume), else the device number of <d>
    through /sys/dev/block (a partition's parent); the first with a /sys/block/<name>/queue wins."""
    names = []
    if source:
        try:
            dev = json.loads(run(["lsblk", "--json", "--nodeps", "--output", "PKNAME,KNAME",
                                  re.sub(r"\[.*$", "", source)]))["blockdevices"][0]
            names += [dev.get("pkname"), dev.get("kname")]
        except (TypeError, ValueError, KeyError, IndexError):
            pass
    try:
        st = os.stat(d)
        link = "/sys/dev/block/%d:%d" % (os.major(st.st_dev), os.minor(st.st_dev))
        if os.path.exists(link):
            real = os.path.realpath(link)
            part = os.path.exists(os.path.join(real, "partition"))
            names.append(os.path.basename(os.path.dirname(real) if part else real))
    except OSError:
        pass
    names = [n for n in names if n]
    for n in names:
        if os.path.isdir("/sys/block/%s/queue" % n):
            return n
    return names[0] if names else None


def docker_bytes(v):
    """docker's --memory value in bytes (k/m/g/t/p are powers of 1024)."""
    m = re.match(r"^([0-9]+(?:\.[0-9]+)?) ?([kmgtp]?)i?b?$", v, re.I)
    return int(float(m.group(1)) * 1024 ** "_kmgtp".index(m.group(2).lower() or "_")) if m else None


def container_memory(start):
    """The --memory (-m) of the docker run line in the node's start script: what HostConfig.Memory holds once
    the unit starts it; 0 when the line has none; None without a readable start script or docker run line."""
    lines = [l for l in (text(start) or "").splitlines()
             if re.search(r"(^|[\s/])docker run\s", l) and not l.lstrip().startswith("#")]
    if not lines:
        return None
    toks = shlex.split(lines[-1])
    for k, tok in enumerate(toks):
        if tok in ("--memory", "-m") and k + 1 < len(toks):
            return docker_bytes(toks[k + 1])
        if tok.startswith("--memory="):
            return docker_bytes(tok[len("--memory="):])
    return 0


def field(f, *a):
    try:
        return f(*a)
    except Exception:
        return None


def probe(d, start):
    h = {"kernel": text("/proc/sys/kernel/osrelease"), "cmdline": text("/proc/cmdline"),
         "os_release": field(key_value, "/etc/os-release", "PRETTY_NAME", "="),
         "image_serial": field(key_value, "/etc/cloud/build.info", "serial", ":"),
         "mem_total_kb": field(mem_total_kb), "nproc": field(lambda: len(os.sched_getaffinity(0))),
         "datadir": d}
    src, fstype, opts = field(mount_of, d) or (None, None, None)
    disk = field(data_disk, d, src)
    h.update([("datadir_mount_source", src), ("datadir_fstype", fstype), ("datadir_mount_options", opts),
              ("disk", disk)])
    q = "/sys/block/%s/queue/" % disk
    for k in ("read_ahead_kb", "max_sectors_kb", "max_hw_sectors_kb", "scheduler", "nr_requests"):
        h[k] = None if disk is None else active(q + k) if k == "scheduler" else number(q + k)
    for k in ("enabled", "defrag", "shmem_enabled"):
        h["thp_" + k] = active(THP + k)
    h["khugepaged_defrag"] = active(THP + "khugepaged/defrag")
    for k in VM:
        h["vm_" + k] = number("/proc/sys/vm/" + k)
    h["container_memory_limit_bytes"] = field(container_memory, start)
    path = os.path.join(d, "db", "mdbx.dat")
    exists = os.path.isfile(path)
    h["mdbx_dat_size_bytes"] = field(lambda: os.stat(path).st_size) if exists else None
    h["mdbx_dat_resident_pages"] = field(resident_pages, path) if exists else None
    h["mdbx_dat_folios"] = None   # no process maps mdbx.dat before the start (see above)
    h["lru_gen_enabled"] = text("/sys/kernel/mm/lru_gen/enabled")
    return h


def evict(d):
    """Resident pages of <d>/db/mdbx.dat before and after fsync + posix_fadvise(DONTNEED); (0, 0) when the
    file is not there (a node that never started has nothing of it cached)."""
    path = os.path.join(d, "db", "mdbx.dat")
    if not os.path.exists(path):
        return 0, 0
    before = resident_pages(path)
    fd = os.open(path, os.O_RDONLY)
    try:
        os.fsync(fd)
        os.posix_fadvise(fd, 0, 0, os.POSIX_FADV_DONTNEED)
    finally:
        os.close(fd)
    after = resident_pages(path)
    if after:
        raise StepFailed("%s keeps %d resident page(s) after fsync and posix_fadvise DONTNEED (%d before)"
                         % (path, after, before))
    return before, after


def read_ahead(d, kb):
    """queue/read_ahead_kb of the disk under <d> before and after writing <kb> to it."""
    disk = data_disk(d, mount_of(d)[0])
    if not disk:
        raise StepFailed("no block device found under %s" % d)
    q = "/sys/block/%s/queue/read_ahead_kb" % disk
    before = number(q)
    if before is None:
        raise StepFailed("cannot read %s" % q)
    try:
        with open(q, "w") as f:
            f.write("%d\n" % kb)
    except OSError as e:
        raise StepFailed("cannot write %d to %s: %s" % (kb, q, why(e)))
    after = number(q)
    if after != kb:
        raise StepFailed("%s reads %s after writing %d (it was %d; the kernel keeps whole pages)" % (q, after, kb, before))
    return before, after


def expired(signum, frame):
    raise RuntimeError("not done after 30 s")


def main(argv):
    mode = argv[1] if len(argv) > 1 else ""
    signal.signal(signal.SIGALRM, expired)
    signal.alarm(30)
    try:
        if mode == "probe":
            try:
                line = "%s %s" % (argv[2], json.dumps(probe(argv[3], argv[4]), separators=(",", ":")))
            except Exception as e:
                line = "%s probe_failed %s" % (argv[2], why(e))
            print("HOST_PROBE " + line)
            return
        try:
            if mode == "evict-mdbx":
                print("ok %d %d" % evict(argv[2]))
            elif mode == "read-ahead-kb":
                print("ok %d %d" % read_ahead(argv[2], int(argv[3])))
            else:
                print("failed unknown mode '%s'" % mode)
        except StepFailed as e:
            print("failed %s" % e)
        except Exception as e:
            print("failed %s" % why(e))
    finally:
        signal.alarm(0)

class Mapped(object):
    """<path> mapped shared and read-only. Mapping faults no page in, and nothing reads the map."""

    def __init__(self, path):
        self.addr = None
        self.fd = os.open(path, os.O_RDONLY)
        self.size = os.fstat(self.fd).st_size
        self.pages = (self.size + PS - 1) // PS
        if self.size:
            c = libc()
            a = c.mmap(None, self.size, mmap.PROT_READ, mmap.MAP_SHARED, self.fd, 0)
            if not a or a == ctypes.c_void_p(-1).value:
                e = ctypes.get_errno()
                self.close()
                raise OSError(e, "mmap: " + os.strerror(e))
            self.addr = a

    def resident(self):
        """mincore of the map: one byte per page, 1 when the page cache holds the page."""
        vec = (ctypes.c_ubyte * max(self.pages, 1))()
        if self.pages and libc().mincore(self.addr, self.size, vec) != 0:
            e = ctypes.get_errno()
            raise OSError(e, "mincore: " + os.strerror(e))
        return bytes(vec)[:self.pages].translate(LOWBIT)

    def close(self):
        if self.addr:
            libc().munmap(self.addr, self.size)
            self.addr = None
        if self.fd is not None:
            os.close(self.fd)
            self.fd = None


def resident_pages(path):
    """Resident pages of <path>: fincore's count when fincore runs, else mincore over an untouched map."""
    out = (run(["fincore", "--noheadings", "--raw", "--output", "PAGES", path]) or "").strip()
    if re.match(r"^[0-9]+$", out):
        return int(out)
    m = Mapped(path)
    try:
        return m.resident().count(1)
    finally:
        m.close()


def mount_of(d):
    """(source, fstype, options) of the mount holding <d>, from findmnt."""
    try:
        fs = json.loads(run(["findmnt", "--json", "--output", "SOURCE,FSTYPE,OPTIONS", "--target", d]))
        fs = fs["filesystems"][0]
        return fs.get("source"), fs.get("fstype"), fs.get("options")
    except (TypeError, ValueError, KeyError, IndexError):
        return None, None, None


def data_disk(d, source):
    """The block device whose queue serves <d>: lsblk's PKNAME of the mount source (the disk under a
    partition), else its KNAME (a whole disk, a device-mapper volume), else the device number of <d>
    through /sys/dev/block (a partition's parent); the first with a /sys/block/<name>/queue wins."""
    names = []
    if source:
        try:
            dev = json.loads(run(["lsblk", "--json", "--nodeps", "--output", "PKNAME,KNAME",
                                  re.sub(r"\[.*$", "", source)]))["blockdevices"][0]
            names += [dev.get("pkname"), dev.get("kname")]
        except (TypeError, ValueError, KeyError, IndexError):
            pass
    try:
        st = os.stat(d)
        link = "/sys/dev/block/%d:%d" % (os.major(st.st_dev), os.minor(st.st_dev))
        if os.path.exists(link):
            real = os.path.realpath(link)
            part = os.path.exists(os.path.join(real, "partition"))
            names.append(os.path.basename(os.path.dirname(real) if part else real))
    except OSError:
        pass
    names = [n for n in names if n]
    for n in names:
        if os.path.isdir("/sys/block/%s/queue" % n):
            return n
    return names[0] if names else None


def docker_bytes(v):
    """docker's --memory value in bytes (k/m/g/t/p are powers of 1024)."""
    m = re.match(r"^([0-9]+(?:\.[0-9]+)?) ?([kmgtp]?)i?b?$", v, re.I)
    return int(float(m.group(1)) * 1024 ** "_kmgtp".index(m.group(2).lower() or "_")) if m else None


def container_memory(start):
    """The --memory (-m) of the docker run line in the node's start script: what HostConfig.Memory holds once
    the unit starts it; 0 when the line has none; None without a readable start script or docker run line."""
    lines = [l for l in (text(start) or "").splitlines()
             if re.search(r"(^|[\s/])docker run\s", l) and not l.lstrip().startswith("#")]
    if not lines:
        return None
    toks = shlex.split(lines[-1])
    for k, tok in enumerate(toks):
        if tok in ("--memory", "-m") and k + 1 < len(toks):
            return docker_bytes(toks[k + 1])
        if tok.startswith("--memory="):
            return docker_bytes(tok[len("--memory="):])
    return 0

def probe(d, start):
    h = {"kernel": text("/proc/sys/kernel/osrelease"), "cmdline": text("/proc/cmdline"),
         "os_release": field(key_value, "/etc/os-release", "PRETTY_NAME", "="),
         "image_serial": field(key_value, "/etc/cloud/build.info", "serial", ":"),
         "mem_total_kb": field(mem_total_kb), "nproc": field(lambda: len(os.sched_getaffinity(0))),
         "datadir": d}
    src, fstype, opts = field(mount_of, d) or (None, None, None)
    disk = field(data_disk, d, src)
    h.update([("datadir_mount_source", src), ("datadir_fstype", fstype), ("datadir_mount_options", opts),
              ("disk", disk)])
    q = "/sys/block/%s/queue/" % disk
    for k in ("read_ahead_kb", "max_sectors_kb", "max_hw_sectors_kb", "scheduler", "nr_requests"):
        h[k] = None if disk is None else active(q + k) if k == "scheduler" else number(q + k)
    for k in ("enabled", "defrag", "shmem_enabled"):
        h["thp_" + k] = active(THP + k)
    h["khugepaged_defrag"] = active(THP + "khugepaged/defrag")
    for k in VM:
        h["vm_" + k] = number("/proc/sys/vm/" + k)
    h["container_memory_limit_bytes"] = field(container_memory, start)
    path = os.path.join(d, "db", "mdbx.dat")
    exists = os.path.isfile(path)
    h["mdbx_dat_size_bytes"] = field(lambda: os.stat(path).st_size) if exists else None
    h["mdbx_dat_resident_pages"] = field(resident_pages, path) if exists else None
    h["mdbx_dat_folios"] = None   # no process maps mdbx.dat before the start (see above)
    h["lru_gen_enabled"] = text("/sys/kernel/mm/lru_gen/enabled")
    return h


def evict(d):
    """Resident pages of <d>/db/mdbx.dat before and after fsync + posix_fadvise(DONTNEED); (0, 0) when the
    file is not there (a node that never started has nothing of it cached)."""
    path = os.path.join(d, "db", "mdbx.dat")
    if not os.path.exists(path):
        return 0, 0
    before = resident_pages(path)
    fd = os.open(path, os.O_RDONLY)
    try:
        os.fsync(fd)
        os.posix_fadvise(fd, 0, 0, os.POSIX_FADV_DONTNEED)
    finally:
        os.close(fd)
    after = resident_pages(path)
    if after:
        raise StepFailed("%s keeps %d resident page(s) after fsync and posix_fadvise DONTNEED (%d before)"
                         % (path, after, before))
    return before, after


def read_ahead(d, kb):
    """queue/read_ahead_kb of the disk under <d> before and after writing <kb> to it."""
    disk = data_disk(d, mount_of(d)[0])
    if not disk:
        raise StepFailed("no block device found under %s" % d)
    q = "/sys/block/%s/queue/read_ahead_kb" % disk
    before = number(q)
    if before is None:
        raise StepFailed("cannot read %s" % q)
    try:
        with open(q, "w") as f:
            f.write("%d\n" % kb)
    except OSError as e:
        raise StepFailed("cannot write %d to %s: %s" % (kb, q, why(e)))
    after = number(q)
    if after != kb:
        raise StepFailed("%s reads %s after writing %d (it was %d; the kernel keeps whole pages)" % (q, after, kb, before))
    return before, after


def expired(signum, frame):
    raise RuntimeError("not done after 30 s")


def main(argv):
    mode = argv[1] if len(argv) > 1 else ""
    signal.signal(signal.SIGALRM, expired)
    signal.alarm(30)
    try:
        if mode == "probe":
            try:
                line = "%s %s" % (argv[2], json.dumps(probe(argv[3], argv[4]), separators=(",", ":")))
            except Exception as e:
                line = "%s probe_failed %s" % (argv[2], why(e))
            print("HOST_PROBE " + line)
            return
        try:
            if mode == "evict-mdbx":
                print("ok %d %d" % evict(argv[2]))
            elif mode == "read-ahead-kb":
                print("ok %d %d" % read_ahead(argv[2], int(argv[3])))
            else:
                print("failed unknown mode '%s'" % mode)
        except StepFailed as e:
            print("failed %s" % e)
        except Exception as e:
            print("failed %s" % why(e))
    finally:
        signal.alarm(0)

def evict(d):
    """Resident pages of <d>/db/mdbx.dat before and after fsync + posix_fadvise(DONTNEED); (0, 0) when the
    file is not there (a node that never started has nothing of it cached)."""
    path = os.path.join(d, "db", "mdbx.dat")
    if not os.path.exists(path):
        return 0, 0
    before = resident_pages(path)
    fd = os.open(path, os.O_RDONLY)
    try:
        os.fsync(fd)
        os.posix_fadvise(fd, 0, 0, os.POSIX_FADV_DONTNEED)
    finally:
        os.close(fd)
    after = resident_pages(path)
    if after:
        raise StepFailed("%s keeps %d resident page(s) after fsync and posix_fadvise DONTNEED (%d before)"
                         % (path, after, before))
    return before, after


def read_ahead(d, kb):
    """queue/read_ahead_kb of the disk under <d> before and after writing <kb> to it."""
    disk = data_disk(d, mount_of(d)[0])
    if not disk:
        raise StepFailed("no block device found under %s" % d)
    q = "/sys/block/%s/queue/read_ahead_kb" % disk
    before = number(q)
    if before is None:
        raise StepFailed("cannot read %s" % q)
    try:
        with open(q, "w") as f:
            f.write("%d\n" % kb)
    except OSError as e:
        raise StepFailed("cannot write %d to %s: %s" % (kb, q, why(e)))
    after = number(q)
    if after != kb:
        raise StepFailed("%s reads %s after writing %d (it was %d; the kernel keeps whole pages)" % (q, after, kb, before))
    return before, after

def libc():

        c.munmap.argtypes = (ctypes.c_void_p, ctypes.c_size_t)
        c.mincore.argtypes = (ctypes.c_void_p, ctypes.c_size_t, ctypes.c_void_p)

class Mapped(object):
    """<path> mapped shared and read-only. Mapping faults no page in, and nothing reads the map."""

    def __init__(self, path):
        self.addr = None
        self.fd = os.open(path, os.O_RDONLY)
        self.size = os.fstat(self.fd).st_size
        self.pages = (self.size + PS - 1) // PS
        if self.size:
            c = libc()
            a = c.mmap(None, self.size, mmap.PROT_READ, mmap.MAP_SHARED, self.fd, 0)
            if not a or a == ctypes.c_void_p(-1).value:
                e = ctypes.get_errno()
                self.close()
                raise OSError(e, "mmap: " + os.strerror(e))
            self.addr = a

    def resident(self):
        """mincore of the map: one byte per page, 1 when the page cache holds the page."""
        vec = (ctypes.c_ubyte * max(self.pages, 1))()
        if self.pages and libc().mincore(self.addr, self.size, vec) != 0:
            e = ctypes.get_errno()
            raise OSError(e, "mincore: " + os.strerror(e))
        return bytes(vec)[:self.pages].translate(LOWBIT)

    def close(self):
        if self.addr:
            libc().munmap(self.addr, self.size)
            self.addr = None
        if self.fd is not None:
            os.close(self.fd)
            self.fd = None


def resident_pages(path):
    """Resident pages of <path>: fincore's count when fincore runs, else mincore over an untouched map."""
    out = (run(["fincore", "--noheadings", "--raw", "--output", "PAGES", path]) or "").strip()
    if re.match(r"^[0-9]+$", out):
        return int(out)
    m = Mapped(path)
    try:
        return m.resident().count(1)
    finally:
        m.close()

def data_disk(d, source):
    """The block device whose queue serves <d>: lsblk's PKNAME of the mount source (the disk under a
    partition), else its KNAME (a whole disk, a device-mapper volume), else the device number of <d>
    through /sys/dev/block (a partition's parent); the first with a /sys/block/<name>/queue wins."""
    names = []
    if source:
        try:
            dev = json.loads(run(["lsblk", "--json", "--nodeps", "--output", "PKNAME,KNAME",
                                  re.sub(r"\[.*$", "", source)]))["blockdevices"][0]
            names += [dev.get("pkname"), dev.get("kname")]
        except (TypeError, ValueError, KeyError, IndexError):
            pass
    try:
        st = os.stat(d)
        link = "/sys/dev/block/%d:%d" % (os.major(st.st_dev), os.minor(st.st_dev))
        if os.path.exists(link):
            real = os.path.realpath(link)
            part = os.path.exists(os.path.join(real, "partition"))
            names.append(os.path.basename(os.path.dirname(real) if part else real))
    except OSError:
        pass
    names = [n for n in names if n]
    for n in names:
        if os.path.isdir("/sys/block/%s/queue" % n):
            return n
    return names[0] if names else None


def docker_bytes(v):
    """docker's --memory value in bytes (k/m/g/t/p are powers of 1024)."""
    m = re.match(r"^([0-9]+(?:\.[0-9]+)?) ?([kmgtp]?)i?b?$", v, re.I)
    return int(float(m.group(1)) * 1024 ** "_kmgtp".index(m.group(2).lower() or "_")) if m else None


def container_memory(start):
    """The --memory (-m) of the docker run line in the node's start script: what HostConfig.Memory holds once
    the unit starts it; 0 when the line has none; None without a readable start script or docker run line."""
    lines = [l for l in (text(start) or "").splitlines()
             if re.search(r"(^|[\s/])docker run\s", l) and not l.lstrip().startswith("#")]
    if not lines:
        return None
    toks = shlex.split(lines[-1])
    for k, tok in enumerate(toks):
        if tok in ("--memory", "-m") and k + 1 < len(toks):
            return docker_bytes(toks[k + 1])
        if tok.startswith("--memory="):
            return docker_bytes(tok[len("--memory="):])
    return 0

def evict(d):
    """Resident pages of <d>/db/mdbx.dat before and after fsync + posix_fadvise(DONTNEED); (0, 0) when the
    file is not there (a node that never started has nothing of it cached)."""
    path = os.path.join(d, "db", "mdbx.dat")
    if not os.path.exists(path):
        return 0, 0
    before = resident_pages(path)
    fd = os.open(path, os.O_RDONLY)
    try:
        os.fsync(fd)
        os.posix_fadvise(fd, 0, 0, os.POSIX_FADV_DONTNEED)
    finally:
        os.close(fd)
    after = resident_pages(path)
    if after:
        raise StepFailed("%s keeps %d resident page(s) after fsync and posix_fadvise DONTNEED (%d before)"
                         % (path, after, before))
    return before, after


def read_ahead(d, kb):
    """queue/read_ahead_kb of the disk under <d> before and after writing <kb> to it."""
    disk = data_disk(d, mount_of(d)[0])
    if not disk:
        raise StepFailed("no block device found under %s" % d)
    q = "/sys/block/%s/queue/read_ahead_kb" % disk
    before = number(q)
    if before is None:
        raise StepFailed("cannot read %s" % q)
    try:
        with open(q, "w") as f:
            f.write("%d\n" % kb)
    except OSError as e:
        raise StepFailed("cannot write %d to %s: %s" % (kb, q, why(e)))
    after = number(q)
    if after != kb:
        raise StepFailed("%s reads %s after writing %d (it was %d; the kernel keeps whole pages)" % (q, after, kb, before))
    return before, after

def why(e):
    return "%s: %s" % (type(e).__name__, " ".join(str(e).split()))


def run(argv):
    """stdout of argv, or None when it is missing, fails or takes over 10 s."""
    try:
        p = subprocess.run(argv, stdin=subprocess.DEVNULL, stdout=subprocess.PIPE, stderr=subprocess.DEVNULL,
                           timeout=10, universal_newlines=True, errors="replace")
    except (OSError, subprocess.SubprocessError):
        return None
    return p.stdout if p.returncode == 0 else None


def text(path):
    try:
        with open(path, errors="replace") as f:
            return f.read().strip()
    except OSError:
        return None


def number(path):
    t = text(path)
    return int(t) if t is not None and re.match(r"^[0-9]+$", t) else None


def active(path):
    """The active entry of a sysfs choice file ("always [madvise] never" -> "madvise"), else its text."""
    t = text(path)
    m = re.search(r"\[([^]]*)\]", t or "")
    return m.group(1) if m else t


def key_value(path, key, sep):
    """The value of the first `<key><sep>` line of <path>, shell quotes removed."""
    for line in (text(path) or "").splitlines():
        k, s, v = line.partition(sep)
        if s and k.strip() == key:
            return " ".join(shlex.split(v))
    return None


def mem_total_kb():

def libc():
    if not _libc:
        c = ctypes.CDLL(None, use_errno=True)
        c.mmap.restype = ctypes.c_void_p
        c.mmap.argtypes = (ctypes.c_void_p, ctypes.c_size_t, ctypes.c_int, ctypes.c_int, ctypes.c_int, ctypes.c_long)
        c.munmap.argtypes = (ctypes.c_void_p, ctypes.c_size_t)
        c.mincore.argtypes = (ctypes.c_void_p, ctypes.c_size_t, ctypes.c_void_p)
        _libc.append(c)
    return _libc[0]


class Mapped(object):
    """<path> mapped shared and read-only. Mapping faults no page in, and nothing reads the map."""

    def __init__(self, path):
        self.addr = None
        self.fd = os.open(path, os.O_RDONLY)
        self.size = os.fstat(self.fd).st_size
        self.pages = (self.size + PS - 1) // PS
        if self.size:
            c = libc()
            a = c.mmap(None, self.size, mmap.PROT_READ, mmap.MAP_SHARED, self.fd, 0)
            if not a or a == ctypes.c_void_p(-1).value:
                e = ctypes.get_errno()
                self.close()
                raise OSError(e, "mmap: " + os.strerror(e))
            self.addr = a

    def resident(self):
        """mincore of the map: one byte per page, 1 when the page cache holds the page."""
        vec = (ctypes.c_ubyte * max(self.pages, 1))()
        if self.pages and libc().mincore(self.addr, self.size, vec) != 0:
            e = ctypes.get_errno()
            raise OSError(e, "mincore: " + os.strerror(e))
        return bytes(vec)[:self.pages].translate(LOWBIT)

    def close(self):
        if self.addr:
            libc().munmap(self.addr, self.size)
            self.addr = None
        if self.fd is not None:
            os.close(self.fd)
            self.fd = None


def resident_pages(path):
    """Resident pages of <path>: fincore's count when fincore runs, else mincore over an untouched map."""
    out = (run(["fincore", "--noheadings", "--raw", "--output", "PAGES", path]) or "").strip()
    if re.match(r"^[0-9]+$", out):
        return int(out)
    m = Mapped(path)
    try:
        return m.resident().count(1)
    finally:
        m.close()


def mount_of(d):
    """(source, fstype, options) of the mount holding <d>, from findmnt."""
    try:
        fs = json.loads(run(["findmnt", "--json", "--output", "SOURCE,FSTYPE,OPTIONS", "--target", d]))
        fs = fs["filesystems"][0]
        return fs.get("source"), fs.get("fstype"), fs.get("options")
    except (TypeError, ValueError, KeyError, IndexError):
        return None, None, None


def data_disk(d, source):
    """The block device whose queue serves <d>: lsblk's PKNAME of the mount source (the disk under a
    partition), else its KNAME (a whole disk, a device-mapper volume), else the device number of <d>
    through /sys/dev/block (a partition's parent); the first with a /sys/block/<name>/queue wins."""
    names = []
    if source:
        try:
            dev = json.loads(run(["lsblk", "--json", "--nodeps", "--output", "PKNAME,KNAME",
                                  re.sub(r"\[.*$", "", source)]))["blockdevices"][0]
            names += [dev.get("pkname"), dev.get("kname")]
        except (TypeError, ValueError, KeyError, IndexError):
            pass
    try:
        st = os.stat(d)
        link = "/sys/dev/block/%d:%d" % (os.major(st.st_dev), os.minor(st.st_dev))
        if os.path.exists(link):
            real = os.path.realpath(link)
            part = os.path.exists(os.path.join(real, "partition"))
            names.append(os.path.basename(os.path.dirname(real) if part else real))
    except OSError:
        pass
    names = [n for n in names if n]
    for n in names:
        if os.path.isdir("/sys/block/%s/queue" % n):
            return n
    return names[0] if names else None


def docker_bytes(v):
    """docker's --memory value in bytes (k/m/g/t/p are powers of 1024)."""
    m = re.match(r"^([0-9]+(?:\.[0-9]+)?) ?([kmgtp]?)i?b?$", v, re.I)
    return int(float(m.group(1)) * 1024 ** "_kmgtp".index(m.group(2).lower() or "_")) if m else None


def container_memory(start):
    """The --memory (-m) of the docker run line in the node's start script: what HostConfig.Memory holds once
    the unit starts it; 0 when the line has none; None without a readable start script or docker run line."""
    lines = [l for l in (text(start) or "").splitlines()
             if re.search(r"(^|[\s/])docker run\s", l) and not l.lstrip().startswith("#")]
    if not lines:
        return None
    toks = shlex.split(lines[-1])
    for k, tok in enumerate(toks):
        if tok in ("--memory", "-m") and k + 1 < len(toks):
            return docker_bytes(toks[k + 1])
        if tok.startswith("--memory="):
            return docker_bytes(tok[len("--memory="):])
    return 0


def field(f, *a):
    try:
        return f(*a)
    except Exception:
        return None


def probe(d, start):
    h = {"kernel": text("/proc/sys/kernel/osrelease"), "cmdline": text("/proc/cmdline"),
         "os_release": field(key_value, "/etc/os-release", "PRETTY_NAME", "="),
         "image_serial": field(key_value, "/etc/cloud/build.info", "serial", ":"),
         "mem_total_kb": field(mem_total_kb), "nproc": field(lambda: len(os.sched_getaffinity(0))),
         "datadir": d}
    src, fstype, opts = field(mount_of, d) or (None, None, None)
    disk = field(data_disk, d, src)
    h.update([("datadir_mount_source", src), ("datadir_fstype", fstype), ("datadir_mount_options", opts),
              ("disk", disk)])
    q = "/sys/block/%s/queue/" % disk
    for k in ("read_ahead_kb", "max_sectors_kb", "max_hw_sectors_kb", "scheduler", "nr_requests"):
        h[k] = None if disk is None else active(q + k) if k == "scheduler" else number(q + k)
    for k in ("enabled", "defrag", "shmem_enabled"):
        h["thp_" + k] = active(THP + k)
    h["khugepaged_defrag"] = active(THP + "khugepaged/defrag")
    for k in VM:
        h["vm_" + k] = number("/proc/sys/vm/" + k)
    h["container_memory_limit_bytes"] = field(container_memory, start)
    path = os.path.join(d, "db", "mdbx.dat")
    exists = os.path.isfile(path)
    h["mdbx_dat_size_bytes"] = field(lambda: os.stat(path).st_size) if exists else None
    h["mdbx_dat_resident_pages"] = field(resident_pages, path) if exists else None
    h["mdbx_dat_folios"] = None   # no process maps mdbx.dat before the start (see above)
    h["lru_gen_enabled"] = text("/sys/kernel/mm/lru_gen/enabled")
    return h


def evict(d):
    """Resident pages of <d>/db/mdbx.dat before and after fsync + posix_fadvise(DONTNEED); (0, 0) when the
    file is not there (a node that never started has nothing of it cached)."""
    path = os.path.join(d, "db", "mdbx.dat")
    if not os.path.exists(path):
        return 0, 0
    before = resident_pages(path)
    fd = os.open(path, os.O_RDONLY)
    try:
        os.fsync(fd)
        os.posix_fadvise(fd, 0, 0, os.POSIX_FADV_DONTNEED)
    finally:
        os.close(fd)
    after = resident_pages(path)
    if after:
        raise StepFailed("%s keeps %d resident page(s) after fsync and posix_fadvise DONTNEED (%d before)"
                         % (path, after, before))
    return before, after


def read_ahead(d, kb):
    """queue/read_ahead_kb of the disk under <d> before and after writing <kb> to it."""
    disk = data_disk(d, mount_of(d)[0])
    if not disk:
        raise StepFailed("no block device found under %s" % d)
    q = "/sys/block/%s/queue/read_ahead_kb" % disk
    before = number(q)
    if before is None:
        raise StepFailed("cannot read %s" % q)
    try:
        with open(q, "w") as f:
            f.write("%d\n" % kb)
    except OSError as e:
        raise StepFailed("cannot write %d to %s: %s" % (kb, q, why(e)))
    after = number(q)
    if after != kb:
        raise StepFailed("%s reads %s after writing %d (it was %d; the kernel keeps whole pages)" % (q, after, kb, before))
    return before, after

def libc():
    if not _libc:
        c = ctypes.CDLL(None, use_errno=True)
        c.mmap.restype = ctypes.c_void_p
        c.mmap.argtypes = (ctypes.c_void_p, ctypes.c_size_t, ctypes.c_int, ctypes.c_int, ctypes.c_int, ctypes.c_long)
        c.munmap.argtypes = (ctypes.c_void_p, ctypes.c_size_t)
        c.mincore.argtypes = (ctypes.c_void_p, ctypes.c_size_t, ctypes.c_void_p)
        _libc.append(c)
    return _libc[0]


class Mapped(object):
    """<path> mapped shared and read-only. Mapping faults no page in, and nothing reads the map."""

    def __init__(self, path):
        self.addr = None
        self.fd = os.open(path, os.O_RDONLY)
        self.size = os.fstat(self.fd).st_size
        self.pages = (self.size + PS - 1) // PS
        if self.size:
            c = libc()
            a = c.mmap(None, self.size, mmap.PROT_READ, mmap.MAP_SHARED, self.fd, 0)
            if not a or a == ctypes.c_void_p(-1).value:
                e = ctypes.get_errno()
                self.close()
                raise OSError(e, "mmap: " + os.strerror(e))
            self.addr = a

    def resident(self):
        """mincore of the map: one byte per page, 1 when the page cache holds the page."""
        vec = (ctypes.c_ubyte * max(self.pages, 1))()
        if self.pages and libc().mincore(self.addr, self.size, vec) != 0:
            e = ctypes.get_errno()
            raise OSError(e, "mincore: " + os.strerror(e))
        return bytes(vec)[:self.pages].translate(LOWBIT)

    def close(self):
        if self.addr:
            libc().munmap(self.addr, self.size)
            self.addr = None
        if self.fd is not None:
            os.close(self.fd)
            self.fd = None


def resident_pages(path):
    """Resident pages of <path>: fincore's count when fincore runs, else mincore over an untouched map."""
    out = (run(["fincore", "--noheadings", "--raw", "--output", "PAGES", path]) or "").strip()
    if re.match(r"^[0-9]+$", out):
        return int(out)
    m = Mapped(path)
    try:
        return m.resident().count(1)
    finally:
        m.close()


def mount_of(d):
    """(source, fstype, options) of the mount holding <d>, from findmnt."""
    try:

def data_disk(d, source):
    """The block device whose queue serves <d>: lsblk's PKNAME of the mount source (the disk under a
    partition), else its KNAME (a whole disk, a device-mapper volume), else the device number of <d>
    through /sys/dev/block (a partition's parent); the first with a /sys/block/<name>/queue wins."""
    names = []
    if source:
        try:
            dev = json.loads(run(["lsblk", "--json", "--nodeps", "--output", "PKNAME,KNAME",
                                  re.sub(r"\[.*$", "", source)]))["blockdevices"][0]
            names += [dev.get("pkname"), dev.get("kname")]
        except (TypeError, ValueError, KeyError, IndexError):
            pass
    try:
        st = os.stat(d)
        link = "/sys/dev/block/%d:%d" % (os.major(st.st_dev), os.minor(st.st_dev))
        if os.path.exists(link):
            real = os.path.realpath(link)
            part = os.path.exists(os.path.join(real, "partition"))
            names.append(os.path.basename(os.path.dirname(real) if part else real))
    except OSError:
        pass
    names = [n for n in names if n]
    for n in names:
        if os.path.isdir("/sys/block/%s/queue" % n):
            return n
    return names[0] if names else None


def docker_bytes(v):
    """docker's --memory value in bytes (k/m/g/t/p are powers of 1024)."""
    m = re.match(r"^([0-9]+(?:\.[0-9]+)?) ?([kmgtp]?)i?b?$", v, re.I)
    return int(float(m.group(1)) * 1024 ** "_kmgtp".index(m.group(2).lower() or "_")) if m else None


def container_memory(start):
    """The --memory (-m) of the docker run line in the node's start script: what HostConfig.Memory holds once
    the unit starts it; 0 when the line has none; None without a readable start script or docker run line."""
    lines = [l for l in (text(start) or "").splitlines()
             if re.search(r"(^|[\s/])docker run\s", l) and not l.lstrip().startswith("#")]
    if not lines:
        return None
    toks = shlex.split(lines[-1])
    for k, tok in enumerate(toks):
        if tok in ("--memory", "-m") and k + 1 < len(toks):
            return docker_bytes(toks[k + 1])
        if tok.startswith("--memory="):
            return docker_bytes(tok[len("--memory="):])
    return 0

def probe(d, start):

         "image_serial": field(key_value, "/etc/cloud/build.info", "serial", ":"),

    h["lru_gen_enabled"] = text("/sys/kernel/mm/lru_gen/enabled")

def evict(d):
    """Resident pages of <d>/db/mdbx.dat before and after fsync + posix_fadvise(DONTNEED); (0, 0) when the
    file is not there (a node that never started has nothing of it cached)."""
    path = os.path.join(d, "db", "mdbx.dat")
    if not os.path.exists(path):
        return 0, 0
    before = resident_pages(path)
    fd = os.open(path, os.O_RDONLY)
    try:
        os.fsync(fd)
        os.posix_fadvise(fd, 0, 0, os.POSIX_FADV_DONTNEED)
    finally:
        os.close(fd)
    after = resident_pages(path)
    if after:
        raise StepFailed("%s keeps %d resident page(s) after fsync and posix_fadvise DONTNEED (%d before)"
                         % (path, after, before))
    return before, after


def read_ahead(d, kb):
    """queue/read_ahead_kb of the disk under <d> before and after writing <kb> to it."""
    disk = data_disk(d, mount_of(d)[0])
    if not disk:
        raise StepFailed("no block device found under %s" % d)
    q = "/sys/block/%s/queue/read_ahead_kb" % disk
    before = number(q)
    if before is None:
        raise StepFailed("cannot read %s" % q)
    try:
        with open(q, "w") as f:
            f.write("%d\n" % kb)
    except OSError as e:
        raise StepFailed("cannot write %d to %s: %s" % (kb, q, why(e)))
    after = number(q)
    if after != kb:
        raise StepFailed("%s reads %s after writing %d (it was %d; the kernel keeps whole pages)" % (q, after, kb, before))
    return before, after

def libc():
    if not _libc:
        c = ctypes.CDLL(None, use_errno=True)
        c.mmap.restype = ctypes.c_void_p
        c.mmap.argtypes = (ctypes.c_void_p, ctypes.c_size_t, ctypes.c_int, ctypes.c_int, ctypes.c_int, ctypes.c_long)
        c.munmap.argtypes = (ctypes.c_void_p, ctypes.c_size_t)
        c.mincore.argtypes = (ctypes.c_void_p, ctypes.c_size_t, ctypes.c_void_p)

    return _libc[0]


class Mapped(object):
    """<path> mapped shared and read-only. Mapping faults no page in, and nothing reads the map."""

    def __init__(self, path):
        self.addr = None
        self.fd = os.open(path, os.O_RDONLY)
        self.size = os.fstat(self.fd).st_size
        self.pages = (self.size + PS - 1) // PS
        if self.size:
            c = libc()
            a = c.mmap(None, self.size, mmap.PROT_READ, mmap.MAP_SHARED, self.fd, 0)
            if not a or a == ctypes.c_void_p(-1).value:
                e = ctypes.get_errno()
                self.close()
                raise OSError(e, "mmap: " + os.strerror(e))
            self.addr = a

    def resident(self):
        """mincore of the map: one byte per page, 1 when the page cache holds the page."""
        vec = (ctypes.c_ubyte * max(self.pages, 1))()
        if self.pages and libc().mincore(self.addr, self.size, vec) != 0:
            e = ctypes.get_errno()
            raise OSError(e, "mincore: " + os.strerror(e))
        return bytes(vec)[:self.pages].translate(LOWBIT)

    def close(self):
        if self.addr:
            libc().munmap(self.addr, self.size)
            self.addr = None
        if self.fd is not None:
            os.close(self.fd)
            self.fd = None


def resident_pages(path):
    """Resident pages of <path>: fincore's count when fincore runs, else mincore over an untouched map."""
    out = (run(["fincore", "--noheadings", "--raw", "--output", "PAGES", path]) or "").strip()
    if re.match(r"^[0-9]+$", out):
        return int(out)
    m = Mapped(path)
    try:
        return m.resident().count(1)
    finally:
        m.close()


def mount_of(d):
    """(source, fstype, options) of the mount holding <d>, from findmnt."""
    try:

def data_disk(d, source):
    """The block device whose queue serves <d>: lsblk's PKNAME of the mount source (the disk under a
    partition), else its KNAME (a whole disk, a device-mapper volume), else the device number of <d>
    through /sys/dev/block (a partition's parent); the first with a /sys/block/<name>/queue wins."""
    names = []
    if source:
        try:
            dev = json.loads(run(["lsblk", "--json", "--nodeps", "--output", "PKNAME,KNAME",
                                  re.sub(r"\[.*$", "", source)]))["blockdevices"][0]
            names += [dev.get("pkname"), dev.get("kname")]
        except (TypeError, ValueError, KeyError, IndexError):
            pass
    try:
        st = os.stat(d)
        link = "/sys/dev/block/%d:%d" % (os.major(st.st_dev), os.minor(st.st_dev))
        if os.path.exists(link):
            real = os.path.realpath(link)
            part = os.path.exists(os.path.join(real, "partition"))
            names.append(os.path.basename(os.path.dirname(real) if part else real))
    except OSError:
        pass
    names = [n for n in names if n]
    for n in names:
        if os.path.isdir("/sys/block/%s/queue" % n):
            return n
    return names[0] if names else None


def docker_bytes(v):
    """docker's --memory value in bytes (k/m/g/t/p are powers of 1024)."""
    m = re.match(r"^([0-9]+(?:\.[0-9]+)?) ?([kmgtp]?)i?b?$", v, re.I)
    return int(float(m.group(1)) * 1024 ** "_kmgtp".index(m.group(2).lower() or "_")) if m else None


def container_memory(start):
    """The --memory (-m) of the docker run line in the node's start script: what HostConfig.Memory holds once
    the unit starts it; 0 when the line has none; None without a readable start script or docker run line."""
    lines = [l for l in (text(start) or "").splitlines()
             if re.search(r"(^|[\s/])docker run\s", l) and not l.lstrip().startswith("#")]
    if not lines:
        return None
    toks = shlex.split(lines[-1])
    for k, tok in enumerate(toks):
        if tok in ("--memory", "-m") and k + 1 < len(toks):
            return docker_bytes(toks[k + 1])
        if tok.startswith("--memory="):
            return docker_bytes(tok[len("--memory="):])
    return 0

def evict(d):
    """Resident pages of <d>/db/mdbx.dat before and after fsync + posix_fadvise(DONTNEED); (0, 0) when the
    file is not there (a node that never started has nothing of it cached)."""
    path = os.path.join(d, "db", "mdbx.dat")
    if not os.path.exists(path):
        return 0, 0
    before = resident_pages(path)
    fd = os.open(path, os.O_RDONLY)
    try:
        os.fsync(fd)
        os.posix_fadvise(fd, 0, 0, os.POSIX_FADV_DONTNEED)
    finally:
        os.close(fd)
    after = resident_pages(path)
    if after:
        raise StepFailed("%s keeps %d resident page(s) after fsync and posix_fadvise DONTNEED (%d before)"
                         % (path, after, before))
    return before, after


def read_ahead(d, kb):
    """queue/read_ahead_kb of the disk under <d> before and after writing <kb> to it."""
    disk = data_disk(d, mount_of(d)[0])
    if not disk:
        raise StepFailed("no block device found under %s" % d)
    q = "/sys/block/%s/queue/read_ahead_kb" % disk
    before = number(q)
    if before is None:
        raise StepFailed("cannot read %s" % q)
    try:
        with open(q, "w") as f:
            f.write("%d\n" % kb)
    except OSError as e:
        raise StepFailed("cannot write %d to %s: %s" % (kb, q, why(e)))
    after = number(q)
    if after != kb:
        raise StepFailed("%s reads %s after writing %d (it was %d; the kernel keeps whole pages)" % (q, after, kb, before))
    return before, after

def libc():
    if not _libc:
        c = ctypes.CDLL(None, use_errno=True)
        c.mmap.restype = ctypes.c_void_p
        c.mmap.argtypes = (ctypes.c_void_p, ctypes.c_size_t, ctypes.c_int, ctypes.c_int, ctypes.c_int, ctypes.c_long)
        c.munmap.argtypes = (ctypes.c_void_p, ctypes.c_size_t)
        c.mincore.argtypes = (ctypes.c_void_p, ctypes.c_size_t, ctypes.c_void_p)

    return _libc[0]


class Mapped(object):
    """<path> mapped shared and read-only. Mapping faults no page in, and nothing reads the map."""

    def __init__(self, path):
        self.addr = None
        self.fd = os.open(path, os.O_RDONLY)
        self.size = os.fstat(self.fd).st_size
        self.pages = (self.size + PS - 1) // PS
        if self.size:
            c = libc()
            a = c.mmap(None, self.size, mmap.PROT_READ, mmap.MAP_SHARED, self.fd, 0)
            if not a or a == ctypes.c_void_p(-1).value:
                e = ctypes.get_errno()
                self.close()
                raise OSError(e, "mmap: " + os.strerror(e))
            self.addr = a

    def resident(self):
        """mincore of the map: one byte per page, 1 when the page cache holds the page."""
        vec = (ctypes.c_ubyte * max(self.pages, 1))()
        if self.pages and libc().mincore(self.addr, self.size, vec) != 0:
            e = ctypes.get_errno()
            raise OSError(e, "mincore: " + os.strerror(e))
        return bytes(vec)[:self.pages].translate(LOWBIT)

    def close(self):
        if self.addr:
            libc().munmap(self.addr, self.size)
            self.addr = None
        if self.fd is not None:
            os.close(self.fd)
            self.fd = None


def resident_pages(path):
    """Resident pages of <path>: fincore's count when fincore runs, else mincore over an untouched map."""
    out = (run(["fincore", "--noheadings", "--raw", "--output", "PAGES", path]) or "").strip()
    if re.match(r"^[0-9]+$", out):
        return int(out)
    m = Mapped(path)
    try:
        return m.resident().count(1)
    finally:
        m.close()

def data_disk(d, source):
    """The block device whose queue serves <d>: lsblk's PKNAME of the mount source (the disk under a
    partition), else its KNAME (a whole disk, a device-mapper volume), else the device number of <d>
    through /sys/dev/block (a partition's parent); the first with a /sys/block/<name>/queue wins."""
    names = []
    if source:
        try:
            dev = json.loads(run(["lsblk", "--json", "--nodeps", "--output", "PKNAME,KNAME",
                                  re.sub(r"\[.*$", "", source)]))["blockdevices"][0]
            names += [dev.get("pkname"), dev.get("kname")]
        except (TypeError, ValueError, KeyError, IndexError):
            pass
    try:
        st = os.stat(d)
        link = "/sys/dev/block/%d:%d" % (os.major(st.st_dev), os.minor(st.st_dev))
        if os.path.exists(link):
            real = os.path.realpath(link)
            part = os.path.exists(os.path.join(real, "partition"))
            names.append(os.path.basename(os.path.dirname(real) if part else real))
    except OSError:
        pass
    names = [n for n in names if n]
    for n in names:
        if os.path.isdir("/sys/block/%s/queue" % n):
            return n
    return names[0] if names else None


def docker_bytes(v):
    """docker's --memory value in bytes (k/m/g/t/p are powers of 1024)."""
    m = re.match(r"^([0-9]+(?:\.[0-9]+)?) ?([kmgtp]?)i?b?$", v, re.I)
    return int(float(m.group(1)) * 1024 ** "_kmgtp".index(m.group(2).lower() or "_")) if m else None


def container_memory(start):
    """The --memory (-m) of the docker run line in the node's start script: what HostConfig.Memory holds once
    the unit starts it; 0 when the line has none; None without a readable start script or docker run line."""
    lines = [l for l in (text(start) or "").splitlines()
             if re.search(r"(^|[\s/])docker run\s", l) and not l.lstrip().startswith("#")]
    if not lines:
        return None
    toks = shlex.split(lines[-1])
    for k, tok in enumerate(toks):
        if tok in ("--memory", "-m") and k + 1 < len(toks):
            return docker_bytes(toks[k + 1])
        if tok.startswith("--memory="):
            return docker_bytes(tok[len("--memory="):])
    return 0

def evict(d):
    """Resident pages of <d>/db/mdbx.dat before and after fsync + posix_fadvise(DONTNEED); (0, 0) when the
    file is not there (a node that never started has nothing of it cached)."""
    path = os.path.join(d, "db", "mdbx.dat")
    if not os.path.exists(path):
        return 0, 0
    before = resident_pages(path)
    fd = os.open(path, os.O_RDONLY)
    try:
        os.fsync(fd)
        os.posix_fadvise(fd, 0, 0, os.POSIX_FADV_DONTNEED)
    finally:
        os.close(fd)
    after = resident_pages(path)
    if after:
        raise StepFailed("%s keeps %d resident page(s) after fsync and posix_fadvise DONTNEED (%d before)"
                         % (path, after, before))
    return before, after


def read_ahead(d, kb):
    """queue/read_ahead_kb of the disk under <d> before and after writing <kb> to it."""
    disk = data_disk(d, mount_of(d)[0])
    if not disk:
        raise StepFailed("no block device found under %s" % d)
    q = "/sys/block/%s/queue/read_ahead_kb" % disk
    before = number(q)
    if before is None:
        raise StepFailed("cannot read %s" % q)
    try:
        with open(q, "w") as f:
            f.write("%d\n" % kb)
    except OSError as e:
        raise StepFailed("cannot write %d to %s: %s" % (kb, q, why(e)))
    after = number(q)
    if after != kb:
        raise StepFailed("%s reads %s after writing %d (it was %d; the kernel keeps whole pages)" % (q, after, kb, before))
    return before, after

def why(e):
    return "%s: %s" % (type(e).__name__, " ".join(str(e).split()))


def run(argv):
    """stdout of argv, or None when it is missing, fails or takes over 10 s."""
    try:
        p = subprocess.run(argv, stdin=subprocess.DEVNULL, stdout=subprocess.PIPE, stderr=subprocess.DEVNULL,
                           timeout=10, universal_newlines=True, errors="replace")
    except (OSError, subprocess.SubprocessError):
        return None
    return p.stdout if p.returncode == 0 else None


def text(path):
    try:
        with open(path, errors="replace") as f:
            return f.read().strip()
    except OSError:
        return None


def number(path):
    t = text(path)
    return int(t) if t is not None and re.match(r"^[0-9]+$", t) else None


def active(path):
    """The active entry of a sysfs choice file ("always [madvise] never" -> "madvise"), else its text."""
    t = text(path)
    m = re.search(r"\[([^]]*)\]", t or "")
    return m.group(1) if m else t


def key_value(path, key, sep):
    """The value of the first `<key><sep>` line of <path>, shell quotes removed."""
    for line in (text(path) or "").splitlines():
        k, s, v = line.partition(sep)
        if s and k.strip() == key:
            return " ".join(shlex.split(v))
    return None


def mem_total_kb():
    for line in (text("/proc/meminfo") or "").splitlines():
        if line.startswith("MemTotal:"):
            return int(line.split()[1])
    return None


def libc():
    if not _libc:
        c = ctypes.CDLL(None, use_errno=True)
        c.mmap.restype = ctypes.c_void_p
        c.mmap.argtypes = (ctypes.c_void_p, ctypes.c_size_t, ctypes.c_int, ctypes.c_int, ctypes.c_int, ctypes.c_long)
        c.munmap.argtypes = (ctypes.c_void_p, ctypes.c_size_t)
        c.mincore.argtypes = (ctypes.c_void_p, ctypes.c_size_t, ctypes.c_void_p)
        _libc.append(c)
    return _libc[0]


class Mapped(object):
    """<path> mapped shared and read-only. Mapping faults no page in, and nothing reads the map."""

    def __init__(self, path):
        self.addr = None
        self.fd = os.open(path, os.O_RDONLY)
        self.size = os.fstat(self.fd).st_size
        self.pages = (self.size + PS - 1) // PS
        if self.size:
            c = libc()
            a = c.mmap(None, self.size, mmap.PROT_READ, mmap.MAP_SHARED, self.fd, 0)
            if not a or a == ctypes.c_void_p(-1).value:
                e = ctypes.get_errno()
                self.close()
                raise OSError(e, "mmap: " + os.strerror(e))
            self.addr = a

    def resident(self):
        """mincore of the map: one byte per page, 1 when the page cache holds the page."""
        vec = (ctypes.c_ubyte * max(self.pages, 1))()
        if self.pages and libc().mincore(self.addr, self.size, vec) != 0:
            e = ctypes.get_errno()
            raise OSError(e, "mincore: " + os.strerror(e))
        return bytes(vec)[:self.pages].translate(LOWBIT)

    def close(self):
        if self.addr:
            libc().munmap(self.addr, self.size)
            self.addr = None
        if self.fd is not None:
            os.close(self.fd)
            self.fd = None


def resident_pages(path):
    """Resident pages of <path>: fincore's count when fincore runs, else mincore over an untouched map."""
    out = (run(["fincore", "--noheadings", "--raw", "--output", "PAGES", path]) or "").strip()
    if re.match(r"^[0-9]+$", out):
        return int(out)
    m = Mapped(path)
    try:
        return m.resident().count(1)
    finally:
        m.close()


def mount_of(d):
    """(source, fstype, options) of the mount holding <d>, from findmnt."""
    try:
        fs = json.loads(run(["findmnt", "--json", "--output", "SOURCE,FSTYPE,OPTIONS", "--target", d]))
        fs = fs["filesystems"][0]
        return fs.get("source"), fs.get("fstype"), fs.get("options")
    except (TypeError, ValueError, KeyError, IndexError):
        return None, None, None


def data_disk(d, source):
    """The block device whose queue serves <d>: lsblk's PKNAME of the mount source (the disk under a
    partition), else its KNAME (a whole disk, a device-mapper volume), else the device number of <d>
    through /sys/dev/block (a partition's parent); the first with a /sys/block/<name>/queue wins."""
    names = []
    if source:
        try:
            dev = json.loads(run(["lsblk", "--json", "--nodeps", "--output", "PKNAME,KNAME",
                                  re.sub(r"\[.*$", "", source)]))["blockdevices"][0]
            names += [dev.get("pkname"), dev.get("kname")]
        except (TypeError, ValueError, KeyError, IndexError):
            pass
    try:
        st = os.stat(d)
        link = "/sys/dev/block/%d:%d" % (os.major(st.st_dev), os.minor(st.st_dev))
        if os.path.exists(link):
            real = os.path.realpath(link)
            part = os.path.exists(os.path.join(real, "partition"))
            names.append(os.path.basename(os.path.dirname(real) if part else real))
    except OSError:
        pass
    names = [n for n in names if n]
    for n in names:
        if os.path.isdir("/sys/block/%s/queue" % n):
            return n
    return names[0] if names else None


def docker_bytes(v):
    """docker's --memory value in bytes (k/m/g/t/p are powers of 1024)."""
    m = re.match(r"^([0-9]+(?:\.[0-9]+)?) ?([kmgtp]?)i?b?$", v, re.I)
    return int(float(m.group(1)) * 1024 ** "_kmgtp".index(m.group(2).lower() or "_")) if m else None


def container_memory(start):
    """The --memory (-m) of the docker run line in the node's start script: what HostConfig.Memory holds once
    the unit starts it; 0 when the line has none; None without a readable start script or docker run line."""
    lines = [l for l in (text(start) or "").splitlines()
             if re.search(r"(^|[\s/])docker run\s", l) and not l.lstrip().startswith("#")]
    if not lines:
        return None
    toks = shlex.split(lines[-1])
    for k, tok in enumerate(toks):
        if tok in ("--memory", "-m") and k + 1 < len(toks):
            return docker_bytes(toks[k + 1])
        if tok.startswith("--memory="):
            return docker_bytes(tok[len("--memory="):])
    return 0


def field(f, *a):
    try:
        return f(*a)
    except Exception:
        return None


def probe(d, start):
    h = {"kernel": text("/proc/sys/kernel/osrelease"), "cmdline": text("/proc/cmdline"),
         "os_release": field(key_value, "/etc/os-release", "PRETTY_NAME", "="),
         "image_serial": field(key_value, "/etc/cloud/build.info", "serial", ":"),
         "mem_total_kb": field(mem_total_kb), "nproc": field(lambda: len(os.sched_getaffinity(0))),
         "datadir": d}
    src, fstype, opts = field(mount_of, d) or (None, None, None)
    disk = field(data_disk, d, src)
    h.update([("datadir_mount_source", src), ("datadir_fstype", fstype), ("datadir_mount_options", opts),
              ("disk", disk)])
    q = "/sys/block/%s/queue/" % disk
    for k in ("read_ahead_kb", "max_sectors_kb", "max_hw_sectors_kb", "scheduler", "nr_requests"):
        h[k] = None if disk is None else active(q + k) if k == "scheduler" else number(q + k)
    for k in ("enabled", "defrag", "shmem_enabled"):
        h["thp_" + k] = active(THP + k)
    h["khugepaged_defrag"] = active(THP + "khugepaged/defrag")
    for k in VM:
        h["vm_" + k] = number("/proc/sys/vm/" + k)
    h["container_memory_limit_bytes"] = field(container_memory, start)
    path = os.path.join(d, "db", "mdbx.dat")
    exists = os.path.isfile(path)
    h["mdbx_dat_size_bytes"] = field(lambda: os.stat(path).st_size) if exists else None
    h["mdbx_dat_resident_pages"] = field(resident_pages, path) if exists else None
    h["mdbx_dat_folios"] = None   # no process maps mdbx.dat before the start (see above)
    h["lru_gen_enabled"] = text("/sys/kernel/mm/lru_gen/enabled")
    return h


def evict(d):
    """Resident pages of <d>/db/mdbx.dat before and after fsync + posix_fadvise(DONTNEED); (0, 0) when the
    file is not there (a node that never started has nothing of it cached)."""
    path = os.path.join(d, "db", "mdbx.dat")
    if not os.path.exists(path):
        return 0, 0
    before = resident_pages(path)
    fd = os.open(path, os.O_RDONLY)
    try:
        os.fsync(fd)
        os.posix_fadvise(fd, 0, 0, os.POSIX_FADV_DONTNEED)
    finally:
        os.close(fd)
    after = resident_pages(path)
    if after:
        raise StepFailed("%s keeps %d resident page(s) after fsync and posix_fadvise DONTNEED (%d before)"
                         % (path, after, before))
    return before, after


def read_ahead(d, kb):
    """queue/read_ahead_kb of the disk under <d> before and after writing <kb> to it."""
    disk = data_disk(d, mount_of(d)[0])
    if not disk:
        raise StepFailed("no block device found under %s" % d)
    q = "/sys/block/%s/queue/read_ahead_kb" % disk
    before = number(q)
    if before is None:
        raise StepFailed("cannot read %s" % q)
    try:
        with open(q, "w") as f:
            f.write("%d\n" % kb)
    except OSError as e:
        raise StepFailed("cannot write %d to %s: %s" % (kb, q, why(e)))
    after = number(q)
    if after != kb:
        raise StepFailed("%s reads %s after writing %d (it was %d; the kernel keeps whole pages)" % (q, after, kb, before))
    return before, after

    raise RuntimeError("not done after 30 s")


def main(argv):
    mode = argv[1] if len(argv) > 1 else ""
    signal.signal(signal.SIGALRM, expired)
    signal.alarm(30)
    try:
        if mode == "probe":
            try:
                line = "%s %s" % (argv[2], json.dumps(probe(argv[3], argv[4]), separators=(",", ":")))
            except Exception as e:
                line = "%s probe_failed %s" % (argv[2], why(e))
            print("HOST_PROBE " + line)
            return
        try:
            if mode == "evict-mdbx":
                print("ok %d %d" % evict(argv[2]))
            elif mode == "read-ahead-kb":
                print("ok %d %d" % read_ahead(argv[2], int(argv[3])))
            else:
                print("failed unknown mode '%s'" % mode)
        except StepFailed as e:
            print("failed %s" % e)
        except Exception as e:
            print("failed %s" % why(e))
    finally:
        signal.alarm(0)

def why(e):

def run(argv):

def text(path):
    try:
        with open(path, errors="replace") as f:
            return f.read().strip()
    except OSError:
        return None


def number(path):
    t = text(path)
    return int(t) if t is not None and re.match(r"^[0-9]+$", t) else None


def active(path):
    """The active entry of a sysfs choice file ("always [madvise] never" -> "madvise"), else its text."""
    t = text(path)
    m = re.search(r"\[([^]]*)\]", t or "")
    return m.group(1) if m else t


def key_value(path, key, sep):
    """The value of the first `<key><sep>` line of <path>, shell quotes removed."""
    for line in (text(path) or "").splitlines():
        k, s, v = line.partition(sep)
        if s and k.strip() == key:
            return " ".join(shlex.split(v))
    return None


def mem_total_kb():
    for line in (text("/proc/meminfo") or "").splitlines():
        if line.startswith("MemTotal:"):

    return None


def libc():
    if not _libc:
        c = ctypes.CDLL(None, use_errno=True)
        c.mmap.restype = ctypes.c_void_p
        c.mmap.argtypes = (ctypes.c_void_p, ctypes.c_size_t, ctypes.c_int, ctypes.c_int, ctypes.c_int, ctypes.c_long)
        c.munmap.argtypes = (ctypes.c_void_p, ctypes.c_size_t)
        c.mincore.argtypes = (ctypes.c_void_p, ctypes.c_size_t, ctypes.c_void_p)
        _libc.append(c)
    return _libc[0]


class Mapped(object):
    """<path> mapped shared and read-only. Mapping faults no page in, and nothing reads the map."""

    def __init__(self, path):
        self.addr = None
        self.fd = os.open(path, os.O_RDONLY)
        self.size = os.fstat(self.fd).st_size
        self.pages = (self.size + PS - 1) // PS
        if self.size:
            c = libc()
            a = c.mmap(None, self.size, mmap.PROT_READ, mmap.MAP_SHARED, self.fd, 0)
            if not a or a == ctypes.c_void_p(-1).value:
                e = ctypes.get_errno()
                self.close()
                raise OSError(e, "mmap: " + os.strerror(e))
            self.addr = a

    def resident(self):
        """mincore of the map: one byte per page, 1 when the page cache holds the page."""
        vec = (ctypes.c_ubyte * max(self.pages, 1))()
        if self.pages and libc().mincore(self.addr, self.size, vec) != 0:
            e = ctypes.get_errno()
            raise OSError(e, "mincore: " + os.strerror(e))
        return bytes(vec)[:self.pages].translate(LOWBIT)

    def close(self):
        if self.addr:
            libc().munmap(self.addr, self.size)
            self.addr = None
        if self.fd is not None:
            os.close(self.fd)
            self.fd = None


def resident_pages(path):
    """Resident pages of <path>: fincore's count when fincore runs, else mincore over an untouched map."""
    out = (run(["fincore", "--noheadings", "--raw", "--output", "PAGES", path]) or "").strip()
    if re.match(r"^[0-9]+$", out):
        return int(out)
    m = Mapped(path)
    try:
        return m.resident().count(1)
    finally:
        m.close()


def mount_of(d):
    """(source, fstype, options) of the mount holding <d>, from findmnt."""
    try:
        fs = json.loads(run(["findmnt", "--json", "--output", "SOURCE,FSTYPE,OPTIONS", "--target", d]))
        fs = fs["filesystems"][0]
        return fs.get("source"), fs.get("fstype"), fs.get("options")
    except (TypeError, ValueError, KeyError, IndexError):
        return None, None, None


def data_disk(d, source):
    """The block device whose queue serves <d>: lsblk's PKNAME of the mount source (the disk under a
    partition), else its KNAME (a whole disk, a device-mapper volume), else the device number of <d>
    through /sys/dev/block (a partition's parent); the first with a /sys/block/<name>/queue wins."""
    names = []
    if source:
        try:
            dev = json.loads(run(["lsblk", "--json", "--nodeps", "--output", "PKNAME,KNAME",
                                  re.sub(r"\[.*$", "", source)]))["blockdevices"][0]
            names += [dev.get("pkname"), dev.get("kname")]
        except (TypeError, ValueError, KeyError, IndexError):
            pass
    try:
        st = os.stat(d)
        link = "/sys/dev/block/%d:%d" % (os.major(st.st_dev), os.minor(st.st_dev))
        if os.path.exists(link):
            real = os.path.realpath(link)
            part = os.path.exists(os.path.join(real, "partition"))
            names.append(os.path.basename(os.path.dirname(real) if part else real))
    except OSError:
        pass
    names = [n for n in names if n]
    for n in names:
        if os.path.isdir("/sys/block/%s/queue" % n):
            return n
    return names[0] if names else None


def docker_bytes(v):
    """docker's --memory value in bytes (k/m/g/t/p are powers of 1024)."""
    m = re.match(r"^([0-9]+(?:\.[0-9]+)?) ?([kmgtp]?)i?b?$", v, re.I)
    return int(float(m.group(1)) * 1024 ** "_kmgtp".index(m.group(2).lower() or "_")) if m else None


def container_memory(start):
    """The --memory (-m) of the docker run line in the node's start script: what HostConfig.Memory holds once
    the unit starts it; 0 when the line has none; None without a readable start script or docker run line."""
    lines = [l for l in (text(start) or "").splitlines()
             if re.search(r"(^|[\s/])docker run\s", l) and not l.lstrip().startswith("#")]
    if not lines:
        return None
    toks = shlex.split(lines[-1])
    for k, tok in enumerate(toks):
        if tok in ("--memory", "-m") and k + 1 < len(toks):
            return docker_bytes(toks[k + 1])
        if tok.startswith("--memory="):
            return docker_bytes(tok[len("--memory="):])
    return 0


def field(f, *a):
    try:
        return f(*a)
    except Exception:
        return None


def probe(d, start):
    h = {"kernel": text("/proc/sys/kernel/osrelease"), "cmdline": text("/proc/cmdline"),
         "os_release": field(key_value, "/etc/os-release", "PRETTY_NAME", "="),
         "image_serial": field(key_value, "/etc/cloud/build.info", "serial", ":"),
         "mem_total_kb": field(mem_total_kb), "nproc": field(lambda: len(os.sched_getaffinity(0))),
         "datadir": d}
    src, fstype, opts = field(mount_of, d) or (None, None, None)
    disk = field(data_disk, d, src)
    h.update([("datadir_mount_source", src), ("datadir_fstype", fstype), ("datadir_mount_options", opts),
              ("disk", disk)])
    q = "/sys/block/%s/queue/" % disk
    for k in ("read_ahead_kb", "max_sectors_kb", "max_hw_sectors_kb", "scheduler", "nr_requests"):
        h[k] = None if disk is None else active(q + k) if k == "scheduler" else number(q + k)
    for k in ("enabled", "defrag", "shmem_enabled"):
        h["thp_" + k] = active(THP + k)
    h["khugepaged_defrag"] = active(THP + "khugepaged/defrag")
    for k in VM:
        h["vm_" + k] = number("/proc/sys/vm/" + k)
    h["container_memory_limit_bytes"] = field(container_memory, start)
    path = os.path.join(d, "db", "mdbx.dat")
    exists = os.path.isfile(path)
    h["mdbx_dat_size_bytes"] = field(lambda: os.stat(path).st_size) if exists else None
    h["mdbx_dat_resident_pages"] = field(resident_pages, path) if exists else None
    h["mdbx_dat_folios"] = None   # no process maps mdbx.dat before the start (see above)
    h["lru_gen_enabled"] = text("/sys/kernel/mm/lru_gen/enabled")
    return h


def evict(d):
    """Resident pages of <d>/db/mdbx.dat before and after fsync + posix_fadvise(DONTNEED); (0, 0) when the
    file is not there (a node that never started has nothing of it cached)."""
    path = os.path.join(d, "db", "mdbx.dat")
    if not os.path.exists(path):
        return 0, 0
    before = resident_pages(path)
    fd = os.open(path, os.O_RDONLY)
    try:
        os.fsync(fd)
        os.posix_fadvise(fd, 0, 0, os.POSIX_FADV_DONTNEED)
    finally:
        os.close(fd)
    after = resident_pages(path)
    if after:
        raise StepFailed("%s keeps %d resident page(s) after fsync and posix_fadvise DONTNEED (%d before)"
                         % (path, after, before))
    return before, after


def read_ahead(d, kb):
    """queue/read_ahead_kb of the disk under <d> before and after writing <kb> to it."""
    disk = data_disk(d, mount_of(d)[0])
    if not disk:
        raise StepFailed("no block device found under %s" % d)
    q = "/sys/block/%s/queue/read_ahead_kb" % disk
    before = number(q)
    if before is None:
        raise StepFailed("cannot read %s" % q)
    try:
        with open(q, "w") as f:
            f.write("%d\n" % kb)
    except OSError as e:
        raise StepFailed("cannot write %d to %s: %s" % (kb, q, why(e)))
    after = number(q)
    if after != kb:
        raise StepFailed("%s reads %s after writing %d (it was %d; the kernel keeps whole pages)" % (q, after, kb, before))
    return before, after

            except Exception as e:
                line = "%s probe_failed %s" % (argv[2], why(e))
            print("HOST_PROBE " + line)

            else:
                print("failed unknown mode '%s'" % mode)
        except StepFailed as e:
            print("failed %s" % e)
        except Exception as e:
            print("failed %s" % why(e))
    finally:
        signal.alarm(0)

    """The active entry of a sysfs choice file ("always [madvise] never" -> "madvise"), else its text."""
    t = text(path)
    m = re.search(r"\[([^]]*)\]", t or "")
    return m.group(1) if m else t


def key_value(path, key, sep):
    """The value of the first `<key><sep>` line of <path>, shell quotes removed."""
    for line in (text(path) or "").splitlines():
        k, s, v = line.partition(sep)
        if s and k.strip() == key:
            return " ".join(shlex.split(v))
    return None


def mem_total_kb():
    for line in (text("/proc/meminfo") or "").splitlines():
        if line.startswith("MemTotal:"):
            return int(line.split()[1])
    return None


def libc():
    if not _libc:
        c = ctypes.CDLL(None, use_errno=True)
        c.mmap.restype = ctypes.c_void_p
        c.mmap.argtypes = (ctypes.c_void_p, ctypes.c_size_t, ctypes.c_int, ctypes.c_int, ctypes.c_int, ctypes.c_long)
        c.munmap.argtypes = (ctypes.c_void_p, ctypes.c_size_t)
        c.mincore.argtypes = (ctypes.c_void_p, ctypes.c_size_t, ctypes.c_void_p)
        _libc.append(c)
    return _libc[0]


class Mapped(object):
    """<path> mapped shared and read-only. Mapping faults no page in, and nothing reads the map."""

    def __init__(self, path):
        self.addr = None
        self.fd = os.open(path, os.O_RDONLY)
        self.size = os.fstat(self.fd).st_size
        self.pages = (self.size + PS - 1) // PS
        if self.size:
            c = libc()
            a = c.mmap(None, self.size, mmap.PROT_READ, mmap.MAP_SHARED, self.fd, 0)
            if not a or a == ctypes.c_void_p(-1).value:
                e = ctypes.get_errno()
                self.close()
                raise OSError(e, "mmap: " + os.strerror(e))
            self.addr = a

    def resident(self):
        """mincore of the map: one byte per page, 1 when the page cache holds the page."""
        vec = (ctypes.c_ubyte * max(self.pages, 1))()
        if self.pages and libc().mincore(self.addr, self.size, vec) != 0:
            e = ctypes.get_errno()
            raise OSError(e, "mincore: " + os.strerror(e))
        return bytes(vec)[:self.pages].translate(LOWBIT)

    def close(self):
        if self.addr:
            libc().munmap(self.addr, self.size)
            self.addr = None
        if self.fd is not None:
            os.close(self.fd)
            self.fd = None


def resident_pages(path):
    """Resident pages of <path>: fincore's count when fincore runs, else mincore over an untouched map."""
    out = (run(["fincore", "--noheadings", "--raw", "--output", "PAGES", path]) or "").strip()
    if re.match(r"^[0-9]+$", out):
        return int(out)
    m = Mapped(path)
    try:
        return m.resident().count(1)
    finally:
        m.close()


def mount_of(d):
    """(source, fstype, options) of the mount holding <d>, from findmnt."""
    try:
        fs = json.loads(run(["findmnt", "--json", "--output", "SOURCE,FSTYPE,OPTIONS", "--target", d]))
        fs = fs["filesystems"][0]
        return fs.get("source"), fs.get("fstype"), fs.get("options")
    except (TypeError, ValueError, KeyError, IndexError):
        return None, None, None


def data_disk(d, source):
    """The block device whose queue serves <d>: lsblk's PKNAME of the mount source (the disk under a
    partition), else its KNAME (a whole disk, a device-mapper volume), else the device number of <d>
    through /sys/dev/block (a partition's parent); the first with a /sys/block/<name>/queue wins."""
    names = []
    if source:
        try:
            dev = json.loads(run(["lsblk", "--json", "--nodeps", "--output", "PKNAME,KNAME",
                                  re.sub(r"\[.*$", "", source)]))["blockdevices"][0]
            names += [dev.get("pkname"), dev.get("kname")]
        except (TypeError, ValueError, KeyError, IndexError):
            pass
    try:
        st = os.stat(d)
        link = "/sys/dev/block/%d:%d" % (os.major(st.st_dev), os.minor(st.st_dev))
        if os.path.exists(link):
            real = os.path.realpath(link)
            part = os.path.exists(os.path.join(real, "partition"))
            names.append(os.path.basename(os.path.dirname(real) if part else real))
    except OSError:
        pass
    names = [n for n in names if n]
    for n in names:
        if os.path.isdir("/sys/block/%s/queue" % n):
            return n
    return names[0] if names else None


def docker_bytes(v):
    """docker's --memory value in bytes (k/m/g/t/p are powers of 1024)."""
    m = re.match(r"^([0-9]+(?:\.[0-9]+)?) ?([kmgtp]?)i?b?$", v, re.I)
    return int(float(m.group(1)) * 1024 ** "_kmgtp".index(m.group(2).lower() or "_")) if m else None


def container_memory(start):
    """The --memory (-m) of the docker run line in the node's start script: what HostConfig.Memory holds once
    the unit starts it; 0 when the line has none; None without a readable start script or docker run line."""
    lines = [l for l in (text(start) or "").splitlines()
             if re.search(r"(^|[\s/])docker run\s", l) and not l.lstrip().startswith("#")]
    if not lines:
        return None
    toks = shlex.split(lines[-1])
    for k, tok in enumerate(toks):
        if tok in ("--memory", "-m") and k + 1 < len(toks):
            return docker_bytes(toks[k + 1])
        if tok.startswith("--memory="):
            return docker_bytes(tok[len("--memory="):])
    return 0


def field(f, *a):
    try:
        return f(*a)
    except Exception:
        return None


def probe(d, start):

         "os_release": field(key_value, "/etc/os-release", "PRETTY_NAME", "="),
         "image_serial": field(key_value, "/etc/cloud/build.info", "serial", ":"),
         "mem_total_kb": field(mem_total_kb), "nproc": field(lambda: len(os.sched_getaffinity(0))),
         "datadir": d}
    src, fstype, opts = field(mount_of, d) or (None, None, None)
    disk = field(data_disk, d, src)
    h.update([("datadir_mount_source", src), ("datadir_fstype", fstype), ("datadir_mount_options", opts),
              ("disk", disk)])

    for k in ("read_ahead_kb", "max_sectors_kb", "max_hw_sectors_kb", "scheduler", "nr_requests"):
        h[k] = None if disk is None else active(q + k) if k == "scheduler" else number(q + k)

        h["thp_" + k] = active(THP + k)
    h["khugepaged_defrag"] = active(THP + "khugepaged/defrag")
    for k in VM:
        h["vm_" + k] = number("/proc/sys/vm/" + k)
    h["container_memory_limit_bytes"] = field(container_memory, start)
    path = os.path.join(d, "db", "mdbx.dat")
    exists = os.path.isfile(path)
    h["mdbx_dat_size_bytes"] = field(lambda: os.stat(path).st_size) if exists else None
    h["mdbx_dat_resident_pages"] = field(resident_pages, path) if exists else None
    h["mdbx_dat_folios"] = None   # no process maps mdbx.dat before the start (see above)
    h["lru_gen_enabled"] = text("/sys/kernel/mm/lru_gen/enabled")
    return h


def evict(d):
    """Resident pages of <d>/db/mdbx.dat before and after fsync + posix_fadvise(DONTNEED); (0, 0) when the
    file is not there (a node that never started has nothing of it cached)."""
    path = os.path.join(d, "db", "mdbx.dat")
    if not os.path.exists(path):
        return 0, 0
    before = resident_pages(path)
    fd = os.open(path, os.O_RDONLY)
    try:
        os.fsync(fd)
        os.posix_fadvise(fd, 0, 0, os.POSIX_FADV_DONTNEED)
    finally:
        os.close(fd)
    after = resident_pages(path)
    if after:
        raise StepFailed("%s keeps %d resident page(s) after fsync and posix_fadvise DONTNEED (%d before)"
                         % (path, after, before))
    return before, after


def read_ahead(d, kb):
    """queue/read_ahead_kb of the disk under <d> before and after writing <kb> to it."""
    disk = data_disk(d, mount_of(d)[0])
    if not disk:
        raise StepFailed("no block device found under %s" % d)
    q = "/sys/block/%s/queue/read_ahead_kb" % disk
    before = number(q)
    if before is None:
        raise StepFailed("cannot read %s" % q)
    try:
        with open(q, "w") as f:
            f.write("%d\n" % kb)
    except OSError as e:
        raise StepFailed("cannot write %d to %s: %s" % (kb, q, why(e)))
    after = number(q)
    if after != kb:
        raise StepFailed("%s reads %s after writing %d (it was %d; the kernel keeps whole pages)" % (q, after, kb, before))
    return before, after

        if self.size:
            c = libc()
            a = c.mmap(None, self.size, mmap.PROT_READ, mmap.MAP_SHARED, self.fd, 0)
            if not a or a == ctypes.c_void_p(-1).value:
                e = ctypes.get_errno()
                self.close()
                raise OSError(e, "mmap: " + os.strerror(e))
            self.addr = a

    def resident(self):
        """mincore of the map: one byte per page, 1 when the page cache holds the page."""
        vec = (ctypes.c_ubyte * max(self.pages, 1))()
        if self.pages and libc().mincore(self.addr, self.size, vec) != 0:
            e = ctypes.get_errno()
            raise OSError(e, "mincore: " + os.strerror(e))
        return bytes(vec)[:self.pages].translate(LOWBIT)

    def close(self):
        if self.addr:
            libc().munmap(self.addr, self.size)
            self.addr = None
        if self.fd is not None:
            os.close(self.fd)
            self.fd = None


def resident_pages(path):
    """Resident pages of <path>: fincore's count when fincore runs, else mincore over an untouched map."""
    out = (run(["fincore", "--noheadings", "--raw", "--output", "PAGES", path]) or "").strip()
    if re.match(r"^[0-9]+$", out):
        return int(out)
    m = Mapped(path)
    try:
        return m.resident().count(1)
    finally:
        m.close()


def mount_of(d):
    """(source, fstype, options) of the mount holding <d>, from findmnt."""
    try:
        fs = json.loads(run(["findmnt", "--json", "--output", "SOURCE,FSTYPE,OPTIONS", "--target", d]))
        fs = fs["filesystems"][0]
        return fs.get("source"), fs.get("fstype"), fs.get("options")
    except (TypeError, ValueError, KeyError, IndexError):
        return None, None, None


def data_disk(d, source):
    """The block device whose queue serves <d>: lsblk's PKNAME of the mount source (the disk under a
    partition), else its KNAME (a whole disk, a device-mapper volume), else the device number of <d>
    through /sys/dev/block (a partition's parent); the first with a /sys/block/<name>/queue wins."""
    names = []
    if source:
        try:
            dev = json.loads(run(["lsblk", "--json", "--nodeps", "--output", "PKNAME,KNAME",
                                  re.sub(r"\[.*$", "", source)]))["blockdevices"][0]
            names += [dev.get("pkname"), dev.get("kname")]
        except (TypeError, ValueError, KeyError, IndexError):
            pass
    try:
        st = os.stat(d)
        link = "/sys/dev/block/%d:%d" % (os.major(st.st_dev), os.minor(st.st_dev))
        if os.path.exists(link):
            real = os.path.realpath(link)
            part = os.path.exists(os.path.join(real, "partition"))
            names.append(os.path.basename(os.path.dirname(real) if part else real))
    except OSError:
        pass
    names = [n for n in names if n]
    for n in names:
        if os.path.isdir("/sys/block/%s/queue" % n):
            return n
    return names[0] if names else None


def docker_bytes(v):
    """docker's --memory value in bytes (k/m/g/t/p are powers of 1024)."""
    m = re.match(r"^([0-9]+(?:\.[0-9]+)?) ?([kmgtp]?)i?b?$", v, re.I)
    return int(float(m.group(1)) * 1024 ** "_kmgtp".index(m.group(2).lower() or "_")) if m else None


def container_memory(start):
    """The --memory (-m) of the docker run line in the node's start script: what HostConfig.Memory holds once
    the unit starts it; 0 when the line has none; None without a readable start script or docker run line."""
    lines = [l for l in (text(start) or "").splitlines()
             if re.search(r"(^|[\s/])docker run\s", l) and not l.lstrip().startswith("#")]
    if not lines:
        return None
    toks = shlex.split(lines[-1])
    for k, tok in enumerate(toks):
        if tok in ("--memory", "-m") and k + 1 < len(toks):
            return docker_bytes(toks[k + 1])
        if tok.startswith("--memory="):
            return docker_bytes(tok[len("--memory="):])
    return 0


def field(f, *a):
    try:
        return f(*a)
    except Exception:
        return None

def evict(d):
    """Resident pages of <d>/db/mdbx.dat before and after fsync + posix_fadvise(DONTNEED); (0, 0) when the
    file is not there (a node that never started has nothing of it cached)."""
    path = os.path.join(d, "db", "mdbx.dat")
    if not os.path.exists(path):
        return 0, 0
    before = resident_pages(path)
    fd = os.open(path, os.O_RDONLY)
    try:
        os.fsync(fd)
        os.posix_fadvise(fd, 0, 0, os.POSIX_FADV_DONTNEED)
    finally:
        os.close(fd)
    after = resident_pages(path)
    if after:
        raise StepFailed("%s keeps %d resident page(s) after fsync and posix_fadvise DONTNEED (%d before)"
                         % (path, after, before))
    return before, after


def read_ahead(d, kb):
    """queue/read_ahead_kb of the disk under <d> before and after writing <kb> to it."""
    disk = data_disk(d, mount_of(d)[0])
    if not disk:
        raise StepFailed("no block device found under %s" % d)
    q = "/sys/block/%s/queue/read_ahead_kb" % disk
    before = number(q)
    if before is None:
        raise StepFailed("cannot read %s" % q)
    try:
        with open(q, "w") as f:
            f.write("%d\n" % kb)
    except OSError as e:
        raise StepFailed("cannot write %d to %s: %s" % (kb, q, why(e)))
    after = number(q)
    if after != kb:
        raise StepFailed("%s reads %s after writing %d (it was %d; the kernel keeps whole pages)" % (q, after, kb, before))
    return before, after

        c.munmap.argtypes = (ctypes.c_void_p, ctypes.c_size_t)

class Mapped(object):
    """<path> mapped shared and read-only. Mapping faults no page in, and nothing reads the map."""

    def __init__(self, path):
        self.addr = None
        self.fd = os.open(path, os.O_RDONLY)
        self.size = os.fstat(self.fd).st_size
        self.pages = (self.size + PS - 1) // PS
        if self.size:
            c = libc()
            a = c.mmap(None, self.size, mmap.PROT_READ, mmap.MAP_SHARED, self.fd, 0)
            if not a or a == ctypes.c_void_p(-1).value:
                e = ctypes.get_errno()
                self.close()
                raise OSError(e, "mmap: " + os.strerror(e))
            self.addr = a

    def resident(self):
        """mincore of the map: one byte per page, 1 when the page cache holds the page."""
        vec = (ctypes.c_ubyte * max(self.pages, 1))()
        if self.pages and libc().mincore(self.addr, self.size, vec) != 0:
            e = ctypes.get_errno()
            raise OSError(e, "mincore: " + os.strerror(e))
        return bytes(vec)[:self.pages].translate(LOWBIT)

    def close(self):
        if self.addr:
            libc().munmap(self.addr, self.size)
            self.addr = None
        if self.fd is not None:
            os.close(self.fd)
            self.fd = None


def resident_pages(path):

    if source:
        try:
            dev = json.loads(run(["lsblk", "--json", "--nodeps", "--output", "PKNAME,KNAME",
                                  re.sub(r"\[.*$", "", source)]))["blockdevices"][0]
            names += [dev.get("pkname"), dev.get("kname")]
        except (TypeError, ValueError, KeyError, IndexError):
            pass
    try:
        st = os.stat(d)
        link = "/sys/dev/block/%d:%d" % (os.major(st.st_dev), os.minor(st.st_dev))
        if os.path.exists(link):
            real = os.path.realpath(link)
            part = os.path.exists(os.path.join(real, "partition"))
            names.append(os.path.basename(os.path.dirname(real) if part else real))
    except OSError:
        pass
    names = [n for n in names if n]
    for n in names:
        if os.path.isdir("/sys/block/%s/queue" % n):
            return n
    return names[0] if names else None


def docker_bytes(v):
    """docker's --memory value in bytes (k/m/g/t/p are powers of 1024)."""
    m = re.match(r"^([0-9]+(?:\.[0-9]+)?) ?([kmgtp]?)i?b?$", v, re.I)
    return int(float(m.group(1)) * 1024 ** "_kmgtp".index(m.group(2).lower() or "_")) if m else None


def container_memory(start):
    """The --memory (-m) of the docker run line in the node's start script: what HostConfig.Memory holds once
    the unit starts it; 0 when the line has none; None without a readable start script or docker run line."""
    lines = [l for l in (text(start) or "").splitlines()
             if re.search(r"(^|[\s/])docker run\s", l) and not l.lstrip().startswith("#")]
    if not lines:
        return None
    toks = shlex.split(lines[-1])
    for k, tok in enumerate(toks):
        if tok in ("--memory", "-m") and k + 1 < len(toks):
            return docker_bytes(toks[k + 1])
        if tok.startswith("--memory="):
            return docker_bytes(tok[len("--memory="):])
    return 0


def field(f, *a):
    try:
        return f(*a)
    except Exception:
        return None

def evict(d):
    """Resident pages of <d>/db/mdbx.dat before and after fsync + posix_fadvise(DONTNEED); (0, 0) when the
    file is not there (a node that never started has nothing of it cached)."""
    path = os.path.join(d, "db", "mdbx.dat")
    if not os.path.exists(path):
        return 0, 0
    before = resident_pages(path)
    fd = os.open(path, os.O_RDONLY)
    try:
        os.fsync(fd)
        os.posix_fadvise(fd, 0, 0, os.POSIX_FADV_DONTNEED)
    finally:
        os.close(fd)
    after = resident_pages(path)
    if after:
        raise StepFailed("%s keeps %d resident page(s) after fsync and posix_fadvise DONTNEED (%d before)"
                         % (path, after, before))
    return before, after


def read_ahead(d, kb):
    """queue/read_ahead_kb of the disk under <d> before and after writing <kb> to it."""
    disk = data_disk(d, mount_of(d)[0])
    if not disk:
        raise StepFailed("no block device found under %s" % d)
    q = "/sys/block/%s/queue/read_ahead_kb" % disk
    before = number(q)
    if before is None:
        raise StepFailed("cannot read %s" % q)
    try:
        with open(q, "w") as f:
            f.write("%d\n" % kb)
    except OSError as e:
        raise StepFailed("cannot write %d to %s: %s" % (kb, q, why(e)))
    after = number(q)
    if after != kb:
        raise StepFailed("%s reads %s after writing %d (it was %d; the kernel keeps whole pages)" % (q, after, kb, before))
    return before, after

def libc():

        c.mmap.argtypes = (ctypes.c_void_p, ctypes.c_size_t, ctypes.c_int, ctypes.c_int, ctypes.c_int, ctypes.c_long)
        c.munmap.argtypes = (ctypes.c_void_p, ctypes.c_size_t)

class Mapped(object):
    """<path> mapped shared and read-only. Mapping faults no page in, and nothing reads the map."""

    def __init__(self, path):
        self.addr = None
        self.fd = os.open(path, os.O_RDONLY)
        self.size = os.fstat(self.fd).st_size
        self.pages = (self.size + PS - 1) // PS
        if self.size:
            c = libc()
            a = c.mmap(None, self.size, mmap.PROT_READ, mmap.MAP_SHARED, self.fd, 0)
            if not a or a == ctypes.c_void_p(-1).value:
                e = ctypes.get_errno()
                self.close()
                raise OSError(e, "mmap: " + os.strerror(e))
            self.addr = a

    def resident(self):
        """mincore of the map: one byte per page, 1 when the page cache holds the page."""
        vec = (ctypes.c_ubyte * max(self.pages, 1))()
        if self.pages and libc().mincore(self.addr, self.size, vec) != 0:
            e = ctypes.get_errno()
            raise OSError(e, "mincore: " + os.strerror(e))
        return bytes(vec)[:self.pages].translate(LOWBIT)

    def close(self):
        if self.addr:
            libc().munmap(self.addr, self.size)
            self.addr = None
        if self.fd is not None:
            os.close(self.fd)
            self.fd = None


def resident_pages(path):
    """Resident pages of <path>: fincore's count when fincore runs, else mincore over an untouched map."""
    out = (run(["fincore", "--noheadings", "--raw", "--output", "PAGES", path]) or "").strip()
    if re.match(r"^[0-9]+$", out):
        return int(out)
    m = Mapped(path)
    try:
        return m.resident().count(1)
    finally:
        m.close()

def data_disk(d, source):
    """The block device whose queue serves <d>: lsblk's PKNAME of the mount source (the disk under a
    partition), else its KNAME (a whole disk, a device-mapper volume), else the device number of <d>
    through /sys/dev/block (a partition's parent); the first with a /sys/block/<name>/queue wins."""
    names = []
    if source:
        try:
            dev = json.loads(run(["lsblk", "--json", "--nodeps", "--output", "PKNAME,KNAME",
                                  re.sub(r"\[.*$", "", source)]))["blockdevices"][0]
            names += [dev.get("pkname"), dev.get("kname")]
        except (TypeError, ValueError, KeyError, IndexError):
            pass
    try:
        st = os.stat(d)
        link = "/sys/dev/block/%d:%d" % (os.major(st.st_dev), os.minor(st.st_dev))
        if os.path.exists(link):
            real = os.path.realpath(link)
            part = os.path.exists(os.path.join(real, "partition"))
            names.append(os.path.basename(os.path.dirname(real) if part else real))
    except OSError:
        pass
    names = [n for n in names if n]
    for n in names:
        if os.path.isdir("/sys/block/%s/queue" % n):
            return n
    return names[0] if names else None

def probe(d, start):
    h = {"kernel": text("/proc/sys/kernel/osrelease"), "cmdline": text("/proc/cmdline"),
         "os_release": field(key_value, "/etc/os-release", "PRETTY_NAME", "="),
         "image_serial": field(key_value, "/etc/cloud/build.info", "serial", ":"),
         "mem_total_kb": field(mem_total_kb), "nproc": field(lambda: len(os.sched_getaffinity(0))),
         "datadir": d}
    src, fstype, opts = field(mount_of, d) or (None, None, None)
    disk = field(data_disk, d, src)
    h.update([("datadir_mount_source", src), ("datadir_fstype", fstype), ("datadir_mount_options", opts),
              ("disk", disk)])
    q = "/sys/block/%s/queue/" % disk
    for k in ("read_ahead_kb", "max_sectors_kb", "max_hw_sectors_kb", "scheduler", "nr_requests"):
        h[k] = None if disk is None else active(q + k) if k == "scheduler" else number(q + k)
    for k in ("enabled", "defrag", "shmem_enabled"):
        h["thp_" + k] = active(THP + k)
    h["khugepaged_defrag"] = active(THP + "khugepaged/defrag")
    for k in VM:
        h["vm_" + k] = number("/proc/sys/vm/" + k)
    h["container_memory_limit_bytes"] = field(container_memory, start)
    path = os.path.join(d, "db", "mdbx.dat")
    exists = os.path.isfile(path)
    h["mdbx_dat_size_bytes"] = field(lambda: os.stat(path).st_size) if exists else None
    h["mdbx_dat_resident_pages"] = field(resident_pages, path) if exists else None
    h["mdbx_dat_folios"] = None   # no process maps mdbx.dat before the start (see above)
    h["lru_gen_enabled"] = text("/sys/kernel/mm/lru_gen/enabled")
    return h


def evict(d):
    """Resident pages of <d>/db/mdbx.dat before and after fsync + posix_fadvise(DONTNEED); (0, 0) when the
    file is not there (a node that never started has nothing of it cached)."""
    path = os.path.join(d, "db", "mdbx.dat")
    if not os.path.exists(path):
        return 0, 0
    before = resident_pages(path)
    fd = os.open(path, os.O_RDONLY)
    try:
        os.fsync(fd)
        os.posix_fadvise(fd, 0, 0, os.POSIX_FADV_DONTNEED)
    finally:
        os.close(fd)
    after = resident_pages(path)
    if after:
        raise StepFailed("%s keeps %d resident page(s) after fsync and posix_fadvise DONTNEED (%d before)"
                         % (path, after, before))
    return before, after


def read_ahead(d, kb):
    """queue/read_ahead_kb of the disk under <d> before and after writing <kb> to it."""
    disk = data_disk(d, mount_of(d)[0])
    if not disk:
        raise StepFailed("no block device found under %s" % d)
    q = "/sys/block/%s/queue/read_ahead_kb" % disk
    before = number(q)
    if before is None:
        raise StepFailed("cannot read %s" % q)
    try:
        with open(q, "w") as f:
            f.write("%d\n" % kb)
    except OSError as e:
        raise StepFailed("cannot write %d to %s: %s" % (kb, q, why(e)))
    after = number(q)
    if after != kb:
        raise StepFailed("%s reads %s after writing %d (it was %d; the kernel keeps whole pages)" % (q, after, kb, before))
    return before, after

def main(argv):
    mode = argv[1] if len(argv) > 1 else ""
    signal.signal(signal.SIGALRM, expired)
    signal.alarm(30)
    try:
        if mode == "probe":
            try:
                line = "%s %s" % (argv[2], json.dumps(probe(argv[3], argv[4]), separators=(",", ":")))
            except Exception as e:
                line = "%s probe_failed %s" % (argv[2], why(e))
            print("HOST_PROBE " + line)
            return
        try:
            if mode == "evict-mdbx":
                print("ok %d %d" % evict(argv[2]))
            elif mode == "read-ahead-kb":
                print("ok %d %d" % read_ahead(argv[2], int(argv[3])))
            else:
                print("failed unknown mode '%s'" % mode)
        except StepFailed as e:
            print("failed %s" % e)
        except Exception as e:
            print("failed %s" % why(e))
    finally:
        signal.alarm(0)
